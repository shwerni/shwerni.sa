    const results: { id: number; success: boolean; error?: unknown }[] = [];
    const errors = [];
    for (const id of ids) {
      try {
        await updateArticleImageAi(id);
        results.push({ id, success: true });

        results.push({ id, success: false, error });
        errors.push(id);
        console.error(`Failed to update image for article ${id}:`, error);
      }

      await sleep(delayMs);
    }

    return results;
  };
